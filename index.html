/** * 118 Freshmart — Sales Sync * * Reads the Sales tab (the second tab in your Outlet Purchase Submission * workbook) and pushes each row to /api/sales-sync. Run manually OR install * a daily trigger to keep Supabase in sync automatically. * * Setup: * 1. Open the SAME workbook that already has the invoice sheet-sync script * (so we reuse the SHARED_SECRET). * 2. Extensions → Apps Script → "+ Add Script" → paste this file alongside * the existing sheet-sync.gs. * 3. Save. Tab name should match SALES_TAB_NAME below (auto-detect handles * common variants). * 4. Run `syncSales()` once manually — authorise when prompted. * 5. (Optional) Triggers → Add trigger → syncSales → Time-driven → Day timer → 6am. * This will auto-push the previous day's data every morning. */ // NOTE: SHARED_SECRET is declared in 118 Sheet Sync.gs and reused here. // Apps Script shares global scope across all .gs files in the same project, // so do NOT redeclare it here — that would throw "Identifier already declared". // The endpoint deployed at https://freshmart-invoices.vercel.app/api/sales-sync const SALES_ENDPOINT = 'https://freshmart-invoices.vercel.app/api/sales-sync'; // Sales tab name — your workbook has it as the second tab. If renamed, update here. const SALES_TAB_NAME = 'WeeklySales'; // change to the actual tab name if different const OUTLET_FULL_TO_CODE = { 'KARAK': 'KRK', 'LANCHANG': 'LC', 'SIMPANG DURIAN': 'SD', 'SIMPANG PELANGAI': 'SP', 'TAMAN MENTAKAB INDAH': 'TMI', 'TELEMONG': 'TLMG', }; const CATEGORY_HEADERS = [ 'Can Food', 'Cigarette', 'DIY', 'Drinks', 'Dry Food', 'Biscuit', 'Dwi Makmur', 'Frozen', 'Fruits', 'Household', 'Ice Cream', 'Meat', 'Others', 'Seafood', 'Telur', 'Vege', 'Top Up', ]; /** * Main entry point. Read Sales tab and push to Supabase via the webhook. * * @param {number} [daysBack] - If provided, only sync rows with sale_date * newer than (today - daysBack days). Use this from the manual refresh * button to get fast partial syncs. Omit (or pass 0) for full-sheet sync. */ function syncSales(daysBack) { const ss = SpreadsheetApp.getActive(); const sheet = ss.getSheetByName(SALES_TAB_NAME); if (!sheet) { throw new Error(`Tab "${SALES_TAB_NAME}" not found — update SALES_TAB_NAME at the top of this script.`); } const range = sheet.getDataRange(); const values = range.getValues(); if (values.length < 2) { Logger.log('No data rows.'); return; } // Optional date cutoff for partial syncs (manual refresh button uses this) let cutoffStr = null; if (daysBack && daysBack > 0) { const cutoff = new Date(); cutoff.setDate(cutoff.getDate() - Number(daysBack)); cutoffStr = Utilities.formatDate(cutoff, Session.getScriptTimeZone(), 'yyyy-MM-dd'); Logger.log('Partial sync: only rows with sale_date >= ' + cutoffStr); } // Find the header row. The WeeklySales tab uses "✅ Refresh Complete" as // the date column header (not literally "Date"), so we anchor on "Outlet" // and assume the date column is the column immediately to its left. let headerRow = -1; let outletIdx = -1; for (let i = 0; i < Math.min(values.length, 10); i++) { const row = values[i].map(v => String(v).trim()); const oIdx = row.findIndex(c => c === 'Outlet' || c === 'outlet' || c === 'OUTLET'); if (oIdx >= 0) { headerRow = i; outletIdx = oIdx; break; } } if (headerRow < 0) { throw new Error('Could not find header row with "Outlet" column.'); } const headers = values[headerRow].map(v => String(v).trim()); // Date is the column immediately before Outlet const dateIdx = outletIdx - 1; if (dateIdx < 0) { throw new Error('Outlet column is at position 0 — expected a date column to its left.'); } const catIndexes = {}; for (const cat of CATEGORY_HEADERS) { const ix = headers.indexOf(cat); if (ix >= 0) catIndexes[cat] = ix; } // Build payload rows const rows = []; for (let i = headerRow + 1; i < values.length; i++) { const r = values[i]; if (!r[dateIdx] || !r[outletIdx]) continue; // Date — could be a Date object or string let saleDate; if (r[dateIdx] instanceof Date) { saleDate = Utilities.formatDate(r[dateIdx], Session.getScriptTimeZone(), 'yyyy-MM-dd'); } else { const parsed = new Date(r[dateIdx]); if (Number.isNaN(parsed.getTime())) continue; saleDate = Utilities.formatDate(parsed, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } // Skip if before cutoff (partial sync mode) if (cutoffStr && saleDate < cutoffStr) continue; // Outlet — could be code (TMI) or full name (TAMAN MENTAKAB INDAH) let outletCode = String(r[outletIdx]).trim().toUpperCase(); if (OUTLET_FULL_TO_CODE[outletCode]) outletCode = OUTLET_FULL_TO_CODE[outletCode]; const amounts = {}; for (const [cat, ix] of Object.entries(catIndexes)) { const val = r[ix]; if (val === '' || val == null) continue; const num = Number(val); if (!Number.isNaN(num)) amounts[cat] = num; } if (Object.keys(amounts).length === 0) continue; rows.push({ sale_date: saleDate, outlet_code: outletCode, amounts }); } Logger.log(`Sending ${rows.length} rows to backend...`); // POST in chunks of 200 to keep payloads small const CHUNK = 200; let totalInserted = 0; for (let i = 0; i < rows.length; i += CHUNK) { const slice = rows.slice(i, i + CHUNK); const resp = UrlFetchApp.fetch(SALES_ENDPOINT, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ secret: SHARED_SECRET, rows: slice }), muteHttpExceptions: true, }); const code = resp.getResponseCode(); const body = resp.getContentText(); Logger.log(`Chunk ${i / CHUNK + 1}: HTTP ${code}, ${body}`); if (code !== 200) throw new Error(`Chunk ${i / CHUNK + 1} failed: ${body}`); const parsed = JSON.parse(body); totalInserted += parsed.inserted || 0; } Logger.log(`Done. ${totalInserted} rows upserted.`); return totalInserted; } /** * Quick test of the endpoint without uploading the whole sheet. */ function testSalesSyncPing() { const resp = UrlFetchApp.fetch(SALES_ENDPOINT, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ secret: SHARED_SECRET, rows: [{ sale_date: '2026-05-17', outlet_code: 'TMI', amounts: { 'Can Food': 100.0 } }], }), muteHttpExceptions: true, }); Logger.log(resp.getResponseCode() + ': ' + resp.getContentText()); } // ═══════════════════════════════════════════════════════════════════════════ // DAILY AUTO-SYNC TRIGGER // ═══════════════════════════════════════════════════════════════════════════ // Run installDailySalesTrigger() ONCE to set up the daily automatic sync. // After install, syncSales runs every morning around 7am Malaysia time and // pushes any new/edited sales rows to Supabase automatically. // // The sync uses UPSERT on (sale_date, outlet_code, category), so re-running // the same day is safe — it just refreshes existing rows. This means late // entries or corrections to past dates will be picked up automatically each // morning. // ═══════════════════════════════════════════════════════════════════════════ /** * One-time setup: install a daily trigger that runs syncSales every morning. * Safe to re-run — old triggers are removed first so you never get duplicates. * * To change the time: edit the .atHour() value below (0-23, script timezone). * To verify after install: run listAllTriggers(). * To stop: run deleteAllSyncSalesTriggers(). */ function installDailySalesTrigger() { deleteAllSyncSalesTriggers(); // clean up any previous installs ScriptApp.newTrigger('syncSales') .timeBased() .everyDays(1) .atHour(7) // 7am script timezone (set in Project Settings → Time zone) .create(); Logger.log('✅ Daily syncSales trigger installed — runs every day around 7am.'); Logger.log('Verify anytime: Apps Script editor → Triggers (clock icon on left sidebar).'); Logger.log('To check via code: run listAllTriggers().'); } /** * Remove all syncSales triggers. Use this if you want to change schedule * or stop the daily auto-sync entirely. */ function deleteAllSyncSalesTriggers() { const triggers = ScriptApp.getProjectTriggers(); let removed = 0; for (const t of triggers) { if (t.getHandlerFunction() === 'syncSales') { ScriptApp.deleteTrigger(t); removed++; } } Logger.log('Removed ' + removed + ' existing syncSales trigger(s).'); } /** * List all currently installed triggers in this script project. * Run this to verify the daily trigger is active and check what's scheduled. */ function listAllTriggers() { const triggers = ScriptApp.getProjectTriggers(); if (triggers.length === 0) { Logger.log('No triggers installed.'); return; } Logger.log('Current triggers (' + triggers.length + '):'); triggers.forEach((t, i) => { Logger.log( (i + 1) + '. handler=' + t.getHandlerFunction() + ' | type=' + t.getEventType() + ' | source=' + t.getTriggerSource() ); }); } // ═══════════════════════════════════════════════════════════════════════════ // MANUAL REFRESH ENDPOINT (Web App) // ═══════════════════════════════════════════════════════════════════════════ // Lets the P&L dashboard's "Refresh sales" button trigger a sync on demand. // // DEPLOYMENT (one-time setup): // 1. Called by /api/refresh-sales from the dashboard. * Expects body: { secret: SHARED_SECRET, days_back: 14 } * Returns: { ok: true, inserted: N, days_back: N } */ function doPost(e) { let body = {}; try { body = e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {}; } catch (err) { return _json({ ok: false, error: 'invalid_json' }); } if (body.secret !== SHARED_SECRET) { return _json({ ok: false, error: 'unauthorized' }); } const daysBack = body.days_back ? Number(body.days_back) : 14; try { const inserted = syncSales(daysBack); return _json({ ok: true, inserted: inserted || 0, days_back: daysBack }); } catch (err) { return _json({ ok: false, error: String(err && err.message || err) }); } } /** * Optional GET handler for health-check / browser test. * Visit your web app URL in a browser — you'll see this response. */ function doGet() { return _json({ ok: true, service: '118 Freshmart manual refresh', endpoint: 'POST with { secret, days_back }' }); } function _json(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
